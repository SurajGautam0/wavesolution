rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Initial rules to get everything working
    // IMPORTANT: These are for development only, not for production!
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
    
    // For public data that anyone can read
    match /services/{serviceId} {
      allow read: if true;
    }
    
    match /testimonials/{testimonialId} {
      // Anyone can read approved testimonials
      allow read: if true;
    }
    
    // Allow anonymous contact form submissions
    match /contacts/{contactId} {
      allow create: if true;
    }
  }
}
